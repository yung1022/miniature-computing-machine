      u = String(u).trim();

      if (u.indexOf('//') === 0) u = 'https:' + u;

      if (/^http:\/\//i.test(u)) u = 'https://' + u.slice(7);

        u = u.replace(/=s\d+[^/?#]*$/, '=s' + px + '-c').replace(/=w\d+-h\d+[^/?#]*$/, '=s' + px + '-c');

      }

      return u;

    // ---- growth odometer: rolling digit drums, fed by the proxy's /public/growth/:cid ----

    // The proxy reads the number from the private server and hands back just that number.

    // Each digit is a drum (a strip 0-9-0-9) that rolls to its new digit, like a mechanical counter.

    const ODO_H = 1.3;            // drum height in em (matches .odo-drum)

    let odoCols = [];             // left -> right: { el, strip, idx, timer }

    let odoValue = 0;

    let odoSign = '';

    const reduceMotion = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

 

    function odoMakeCol() {

      const el = document.createElement('div');

      el.className = 'odo-drum';

      const strip = document.createElement('div');

      strip.className = 'odo-strip';

      for (let k = 0; k < 20; k++) {

        const d = document.createElement('span');

        d.textContent = String(k % 10);

        strip.appendChild(d);

      }

      el.appendChild(strip);

      return { el: el, strip: strip, idx: 0, timer: 0 };

    }

 

    function odoPlace(col, d) {   // jump, no animation

      clearTimeout(col.timer); col.timer = 0;

      col.strip.style.transition = 'none';

      col.strip.style.transform = 'translateY(' + (-d * ODO_H) + 'em)';

      col.idx = d;

    }

 

    function odoRoll(col, d, up, dur) {

      if (col.timer) { clearTimeout(col.timer); col.timer = 0; odoPlace(col, col.idx); void col.strip.offsetWidth; }

      if (d === col.idx) return;

      if (reduceMotion) return odoPlace(col, d);

      // counting up past 9 keeps rolling forward onto the second 0-9 run, then snaps back unseen

      const wrap = up && d < col.idx;

      const target = wrap ? 10 + d : d;

      col.strip.style.transition = 'transform ' + dur + 'ms cubic-bezier(0.22, 0.8, 0.24, 1)';

      col.strip.style.transform = 'translateY(' + (-target * ODO_H) + 'em)';

      col.idx = d;

      if (wrap) col.timer = setTimeout(() => { col.timer = 0; odoPlace(col, d); }, dur + 40);

    }

 

    // rebuild the row only when the number of digits (or the sign) changes

    function odoLayout(len, sign) {

      if (odoCols.length === len && odoSign === sign && odoTrack.firstChild) return;

      while (odoCols.length < len) odoCols.unshift(odoMakeCol());

      while (odoCols.length > len) odoCols.shift();

      odoTrack.innerHTML = '';

      if (sign) {

        const sg = document.createElement('span');

        sg.className = 'odo-sign';

        sg.textContent = sign;

        odoTrack.appendChild(sg);

      }

      odoCols.forEach((col, i) => {

        if (i > 0 && (len - i) % 3 === 0) {

          const sep = document.createElement('span');

          sep.className = 'odo-sep';

          sep.textContent = ',';

          odoTrack.appendChild(sep);

        }

        odoTrack.appendChild(col.el);

      });

      odoSign = sign;

      odoWindow.classList.toggle('long', len > 9);

      void odoTrack.offsetWidth;

    }

 

    function setOdometer(n, instant) {

      n = Math.round(Number(n) || 0);

      const digits = String(Math.abs(n)).split('');

      const sign = n > 0 ? '+' : (n < 0 ? '\u2212' : '');

      const up = n >= odoValue;

      odoLayout(digits.length, sign);

      odoCols.forEach((col, i) => {

        const d = parseInt(digits[i], 10);

        const changed = d !== col.idx;

        if (instant) { odoPlace(col, d); return; }

        odoRoll(col, d, up, 1300 + (digits.length - 1 - i) * 110);   // left drums settle last

        if (changed) {   // digits that moved glow green (up) or red (down), then fade back to white

          col.el.classList.toggle('odo-up', up);

          col.el.classList.toggle('odo-down', !up);

          clearTimeout(col.tint);

          col.tint = setTimeout(() => col.el.classList.remove('odo-up', 'odo-down'), 3000);

        }

      });

      odoValue = n;

      odo.setAttribute('aria-label', 'Your growth: ' + (n > 0 ? '+' : '') + n.toLocaleString());

    }

    // The proxy answers 429 with {kind: 'cooldown'|'daily', retryAfterSeconds}; the

    // FYSC server's own per-command cooldown answers 429 with plain text ("... 42s ...").

    function parseLimit(text, fallbackSeconds) {

      try {

        const j = JSON.parse(text);

        if (j && j.kind) return { kind: j.kind, seconds: Math.max(1, j.retryAfterSeconds | 0), message: j.message };

      } catch (e) { /* plain text */ }

      const match = text.match(/(\d+)s/);

      const seconds = match ? parseInt(match[1], 10) : fallbackSeconds;

      // "Too many grinds this minute. Try again in 12s." = the server's per-minute cap for the whole channel

      if (/too many grinds|too fast/i.test(text)) return { kind: 'rate', seconds: seconds, message: text };

      return { kind: 'cooldown', seconds: seconds };

    }

    async function doCommand(cmd, btn) {

      if (!selectedChannel || btn.disabled) return;

      btn.classList.remove('punch');

      void btn.offsetWidth; // restart the punch animation on repeat clicks

      btn.classList.add('punch');

      // lock everything right away so a fast double-click can't send two

      btn._busy = true;

      if (!cmd.independent) lockAll(COMMAND_COOLDOWN_SECONDS);

      ensureLockTimer();

      setStatus('Sending ' + cmd.label + '...', 'wait');

      const channel = selectedChannel;

 

      try {

        const response = await fetch(commandUrl(channel.cid), {

          method: 'POST',

          headers: Object.assign({ 'Content-Type': 'application/json' }, authHeaders(channel.cid)),

          body: JSON.stringify(Object.assign({ type: cmd.type, gain: cmd.gain }, cmd.extra || {}))

        });

        const text = await response.text();

        btn._busy = false;

 

        if (response.status === 422) {   // the server refused this one (collab/disstrack: no such channel / on cooldown)

          releaseCommand(cmd);

          setStatus(text || "That didn't work.", 'err');

          const own = text.match(/^Your (?:collab|disstrack) is on cooldown.*?(?:(\d+)m\s*)?(\d+)s/);

          if (own) lockButton(btn, (parseInt(own[1] || '0', 10) * 60) + parseInt(own[2], 10));

          return;

        }

 

        if (response.status === 429) {

          const limit = parseLimit(text, COMMAND_COOLDOWN_SECONDS);

          if (limit.kind === 'rate') {   // the whole channel is going too fast for this minute: everything waits

            lockLimit(limit.seconds);

            setStatus(limit.message || ('Too many grinds this minute. Try again in ' + fmtCountdown(limit.seconds) + '.'), 'wait');

            return;

          }

          if (limit.kind === 'pause') {   // a boost just went through: every other button waits

            lockPause(limit.seconds);

            setStatus(limit.message || ('Commands are paused after a boost. Try again in ' + fmtCountdown(limit.seconds) + '.'), 'wait');

            return;

          }

          if (limit.kind === 'daily') {

            lockLimit(limit.seconds);

            setStatus(limit.message || ('Daily grind limit reached. Try again in ' + fmtCountdown(limit.seconds) + '.'), 'err');

            refreshGrindStatus(channel);

            return;

          }

          setStatus(cmd.label + ' is on cooldown — try again in ' + fmtCountdown(limit.seconds), 'wait');

          // boost only ever waits on itself; for the others, a short wait is the shared

          // cooldown (everything waits) and a long one is that command's own cooldown

          if (cmd.independent) { lockButton(btn, limit.seconds); return; }

          lockAll(Math.min(limit.seconds, COMMAND_COOLDOWN_SECONDS));

          if (limit.seconds > COMMAND_COOLDOWN_SECONDS) lockButton(btn, limit.seconds);

          return;

        }

 

        if (response.status === 409) {   // the server says this channel is on chat grinding

          releaseCommand(cmd);

          setMode('chat');

          setStatus('Chat grinding is on for this channel. Switch the grinding method to Website to use the buttons.', 'err');

          return;

        }

 

        if (response.status === 428) {   // the server wants a quick human check first

          releaseCommand(cmd);

          showCheck(cmd, btn);

          return;

        }

 

        if (response.status === 401 || response.status === 403) {

          clearStoredToken(channel.cid);

          releaseCommand(cmd);

          setStatus('Your login expired. Log in again.', 'err');

          showAuthView(channel);

          return;

        }

 

        if (text.indexOf('not in the system') !== -1) {

          releaseCommand(cmd);

          setStatus("This channel isn't registered yet.", 'err');

          return;

        }

 

        if (!response.ok) {

          releaseCommand(cmd);

          console.error('Command failed: HTTP ' + response.status, text.slice(0, 300));

          setStatus(failureMessage(response.status, text), 'err');

          return;

        }

 

        let result = null;

        try { result = JSON.parse(text); } catch (e) { /* older server: plain text, no amount to show */ }

        if (cmd.type === 'disstrack music video' && result && Number.isFinite(Number(result.loss))) {

          showGain('Disstrack on ' + (result.target || 'that channel') + ' \u2014 they lost', -result.loss);

          const dissMsg = 'Disstrack dropped on ' + (result.target || 'that channel') + '! They lost ' + formatNumber(result.loss) + ' subs.';

          setStatus(dissMsg, 'ok');

          autoClearStatus(dissMsg);

        } else

        if (result && Number.isFinite(Number(result.gain))) {

          if (cmd.type === 'collab') {

            // the server now pays a collab out gradually, so say "gaining" and for how long

            const dripMin = Math.max(1, Math.round((Number(result.dripMs) || 300000) / 60000));

            showGain('Collab with ' + (result.with || 'that channel') + ' \u2014 you both are gaining', result.gain);

            const collabMsg = 'Collab started with ' + (result.with || 'that channel') + '! Both channels will gain +' + formatNumber(result.gain) + 'subs';

            setStatus(collabMsg, 'ok');

            autoClearStatus(collabMsg);

          } else {

            showGain(cmd.label + ' gained', result.gain);

            const okMsg = cmd.label + ' sent! You gained +' + formatNumber(result.gain) + '.';

            setStatus(okMsg, 'ok');

            autoClearStatus(okMsg);

          }

        } else {

          setStatus(cmd.label + ' sent! Watch your live count below.', 'ok');

        }

        if (cmd.type === 'boost') lockPause(BOOST_PAUSE_SECONDS);   // pause every other command for 10s

        else if (!cmd.independent) lockAll(COMMAND_COOLDOWN_SECONDS);

        lockButton(btn, cmd.cooldownSeconds || COMMAND_COOLDOWN_SECONDS);

        refreshGrindStatus(channel);

        refreshGrowth(channel);

      } catch (err) {

        console.error('Command request failed:', err);

        btn._busy = false;

        releaseCommand(cmd);

        setStatus("Couldn't reach the server. Try again.", 'err');

      }

    }

      function renderList() {

        const q = input.value.toLowerCase().trim();

        matches = [];

        for (let i = 0; i < allChannels.length && matches.length < 8; i++) {

        list.innerHTML = '';

        if (!matches.length) {

          const li = document.createElement('li');

          li.className = 'opt-empty';

          li.textContent = 'No channel found';

          list.appendChild(li);

        }

        matches.forEach((c, idx) => {

          const li = document.createElement('li');

          li.setAttribute('role', 'option');

          li.dataset.idx = idx;

          const sb = document.createElement('span'); sb.className = 'opt-subs'; sb.textContent = formatNumber(c.subscriberCountRealAPI) + ' subs';

          li.appendChild(n); li.appendChild(sb);

        active = -1;

        input.setAttribute('aria-expanded', 'true');

        if (target) { target = null; ensureLockTimer(); }   // typing again un-picks the channel

        clearTimeout(timer);

        timer = setTimeout(renderList, 80);

      });

      input.addEventListener('focus', () => { if (!target) renderList(); });

      input.addEventListener('keydown', (e) => {

        if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {

          if (list.style.display !== 'block') renderList();

          e.preventDefault();

          const rowsNow = list.querySelectorAll('li[data-idx]');

          if (!rowsNow.length) return;

          active = (active + (e.key === 'ArrowDown' ? 1 : -1) + rowsNow.length) % rowsNow.length;

          rowsNow.forEach((r, i) => r.classList.toggle('active', i === active));

          rowsNow[active].scrollIntoView({ block: 'nearest' });

        } else if (e.key === 'Enter') {

          e.preventDefault();

          clearTimeout(timer);

          if (list.style.display !== 'block') renderList();

          const choice = matches[active >= 0 ? active : 0];

          if (choice) pick(choice);

        } else if (e.key === 'Escape') {

        }

      });

      });

        if (!box.contains(e.target)) closeList();

      });

 

      return { reset: reset };

    async function pollClaimStatus() {

      if (authMode !== 'claim-wait' || !selectedChannel || !currentClaimToken) return;

      try {

        const response = await fetch(window.FYSC_API_BASE + '/auth/claim/status/' + encodeURIComponent(selectedChannel.cid) +

          '?claimToken=' + encodeURIComponent(currentClaimToken));

        if (!response.ok) throw new Error('HTTP ' + response.status);

        const data = await response.json();

        if (authMode !== 'claim-wait') return; // left this step while the request was in flight

        if (data.verified) {

          setAuthMode('claim-password');

          setAuthStatus('Channel verified.', 'ok');

          return;

        }

        if (data.expired) {

          setAuthMode('claim-start');

          setAuthStatus('That code expired. Get a new one.', 'err');

          return;

        }

      } catch (err) {

        console.error('Claim status check failed:', err);

      }

      claimPollTimer = setTimeout(pollClaimStatus, CLAIM_POLL_MS);

    }

    async function postAuth(path, body) {

      const response = await fetch(window.FYSC_API_BASE + path + '/' + encodeURIComponent(selectedChannel.cid), {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(body)

      });

      const text = await response.text();

      if (response.ok) return JSON.parse(text);

 

      setAuthStatus(text || 'Something went wrong.', 'err');

      authSubmitButton.disabled = false;

      if (authMode === 'login' || authMode === 'claim-start') refreshCaptcha();

      // An expired or already-used claim can't be retried, so start the claim over.

      if (authMode === 'claim-password' && response.status === 403 && /expired/i.test(text)) {

        setAuthMode('claim-start');

        setAuthStatus(text, 'err');

      }

      return null;

    }

    // Mirrors /rename/:cid?to=... (fysca_devasset/server.js), but through

    // the /web/rename/:cid counterpart that requires a login token for this

    // cid. The plain /rename/:cid route only takes the shared command key

    async function doRename() {

      if (!selectedChannel) return;

      const newName = renameInput.value.trim();

      if (!newName) {

        setStatus('Type a new name first.', 'err');

        return;

      }

      // Same limit the server enforces (a letter or emoji counts once). Over it, the server would put the

      // name back to the channel's default, so don't even send it.

      if (Array.from(newName).length > MAX_NAME_LENGTH) {

        setStatus('Names can be at most ' + MAX_NAME_LENGTH + ' characters (yours is ' + Array.from(newName).length + ').', 'err');

        return;

      }

 

      renameButton.disabled = true;

      setStatus('Renaming...', 'wait');

 

      try {

        const url = window.FYSC_API_BASE + '/web/rename/' + encodeURIComponent(selectedChannel.cid) +

          '?to=' + encodeURIComponent(newName);

        const response = await fetch(url, { headers: authHeaders(selectedChannel.cid) });

 

        if (response.status === 401 || response.status === 403) {

          clearStoredToken(selectedChannel.cid);

          setStatus('Your login expired. Log in again.', 'err');

          showAuthView(selectedChannel);

          return;

        }

 

        // 422 = the server refused the name (not allowed, or too long): show its reason, nothing was renamed

        if (response.status === 422) {

          setStatus((await response.text()) || "That name isn't allowed. Pick a different one.", 'err');

          return;

        }

 

        if (!response.ok) throw new Error('HTTP ' + response.status);

 

        // The backend only applies renamedTo for 24-char (standard YouTube)

        // channel IDs -- see /public/s9_data in fysca_devasset/server.js.

        // Reflect that here instead of claiming success either way.

        if (selectedChannel.cid.length !== 24) {

          setStatus("Renamed on the backend, but this channel's ID isn't the usual 24-character format, so the new name may not show up elsewhere.", 'err');

        } else {

          selectedChannel.cname = newName;

          grindName.textContent = newName;

          const inList = allChannels.find(c => c.cid === selectedChannel.cid);

          if (inList) { inList.cname = newName; inList._key = (newName + ' ' + inList.cid).toLowerCase(); }

          setStatus('Renamed to "' + newName + '"!', 'ok');

          renameInput.value = '';

        }

      } catch (err) {

        console.error('Rename request failed:', err);

        setStatus("Couldn't reach the server. Try again.", 'err');

      } finally {

        renameButton.disabled = false;

      }

    }
